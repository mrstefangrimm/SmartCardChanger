```csharp
using System;
using System.Collections.Generic;
using System.Linq;

public class SimpleNeuralNetwork
{
    private double[] weights;
    private double[] biases;
    private Random random = new Random();
    private double learningRate = 0.01;

    // Constructor: initialize network with random weights
    public SimpleNeuralNetwork(int inputSize, int hiddenSize, int outputSize)
    {
        // Layer 1: input -> hidden
        int layer1Weights = inputSize * hiddenSize;
        int layer2Weights = hiddenSize * outputSize;
        
        weights = new double[layer1Weights + layer2Weights];
        biases = new double[hiddenSize + outputSize];
        
        // Random initialization
        for (int i = 0; i < weights.Length; i++)
            weights[i] = (random.NextDouble() - 0.5) * 0.1;
        for (int i = 0; i < biases.Length; i++)
            biases[i] = (random.NextDouble() - 0.5) * 0.1;
    }

    private double ReLU(double x) => Math.Max(0, x);
    private double ReLUDerivative(double x) => x > 0 ? 1 : 0;

    public double Predict(double input)
    {
        // Forward pass (simplified for 1 input)
        double[] hidden = new double[16];
        
        // Input to hidden
        for (int i = 0; i < 16; i++)
        {
            hidden[i] = input * weights[i] + biases[i];
            hidden[i] = ReLU(hidden[i]);
        }
        
        // Hidden to output
        double output = 0;
        for (int i = 0; i < 16; i++)
            output += hidden[i] * weights[16 + i];
        output += biases[16];
        
        return output;
    }

    public void Train(double[][] inputs, double[][] outputs, int epochs)
    {
        for (int epoch = 0; epoch < epochs; epoch++)
        {
            double totalError = 0;
            for (int i = 0; i < inputs.Length; i++)
            {
                double predicted = Predict(inputs[i][0]);
                double error = outputs[i][0] - predicted;
                totalError += error * error;
                
                // Simple gradient update (simplified backprop)
                for (int w = 0; w < weights.Length; w++)
                    weights[w] += learningRate * error * 0.001;
            }
            
            if (epoch % 100 == 0)
                Console.WriteLine($"Epoch {epoch}: MSE = {totalError / inputs.Length}");
        }
    }
}
```

```csharp
using System.Collections.Generic;

public class NeuralNetworkRegression
{
    private double[][] weightsLayer1;  // input -> hidden
    private double[][] weightsLayer2;  // hidden -> output
    private double[] biasesLayer1;
    private double[] biasesLayer2;
    private int inputSize = 1;
    private int hiddenSize = 16;
    private int outputSize = 1;
    private double learningRate = 0.01;
    private Random random = new Random(42);

    public NeuralNetworkRegression()
    {
        InitializeWeights();
    }

    private void InitializeWeights()
    {
        // Layer 1: input (1) -> hidden (16)
        weightsLayer1 = new double[inputSize][];
        for (int i = 0; i < inputSize; i++)
        {
            weightsLayer1[i] = new double[hiddenSize];
            for (int j = 0; j < hiddenSize; j++)
                weightsLayer1[i][j] = (random.NextDouble() - 0.5) * 0.5;
        }

        // Layer 2: hidden (16) -> output (1)
        weightsLayer2 = new double[hiddenSize][];
        for (int i = 0; i < hiddenSize; i++)
        {
            weightsLayer2[i] = new double[outputSize];
            for (int j = 0; j < outputSize; j++)
                weightsLayer2[i][j] = (random.NextDouble() - 0.5) * 0.5;
        }

        biasesLayer1 = new double[hiddenSize];
        biasesLayer2 = new double[outputSize];
    }

    private double ReLU(double x) => Math.Max(0, x);
    private double ReLUDerivative(double x) => x > 0 ? 1 : 0;

    public double Predict(double input)
    {
        // Forward pass
        double[] hiddenInput = new double[hiddenSize];
        double[] hiddenOutput = new double[hiddenSize];
        
        // Input to hidden layer
        for (int j = 0; j < hiddenSize; j++)
        {
            hiddenInput[j] = weightsLayer1[0][j] * input + biasesLayer1[j];
            hiddenOutput[j] = ReLU(hiddenInput[j]);
        }

        // Hidden to output layer
        double output = biasesLayer2[0];
        for (int j = 0; j < hiddenSize; j++)
            output += weightsLayer2[j][0] * hiddenOutput[j];

        return output;
    }

    public void Train(double[] inputs, double[] targets, int epochs)
    {
        for (int epoch = 0; epoch < epochs; epoch++)
        {
            double totalError = 0;

            for (int sampleIdx = 0; sampleIdx < inputs.Length; sampleIdx++)
            {
                double input = inputs[sampleIdx];
                double target = targets[sampleIdx];

                // Forward pass - store intermediate values for backprop
                double[] hiddenInput = new double[hiddenSize];
                double[] hiddenOutput = new double[hiddenSize];

                for (int j = 0; j < hiddenSize; j++)
                {
                    hiddenInput[j] = weightsLayer1[0][j] * input + biasesLayer1[j];
                    hiddenOutput[j] = ReLU(hiddenInput[j]);
                }

                double prediction = biasesLayer2[0];
                for (int j = 0; j < hiddenSize; j++)
                    prediction += weightsLayer2[j][0] * hiddenOutput[j];

                // Compute error
                double error = target - prediction;
                totalError += error * error;

                // Backpropagation
                // Output layer gradients
                double outputGradient = -2 * error;  // derivative of MSE
                
                // Update layer 2 weights and bias
                for (int j = 0; j < hiddenSize; j++)
                {
                    double gradient = outputGradient * hiddenOutput[j];
                    weightsLayer2[j][0] -= learningRate * gradient;
                }
                biasesLayer2[0] -= learningRate * outputGradient;

                // Hidden layer gradients (backprop through ReLU)
                double[] hiddenGradients = new double[hiddenSize];
                for (int j = 0; j < hiddenSize; j++)
                {
                    hiddenGradients[j] = outputGradient * weightsLayer2[j][0] * ReLUDerivative(hiddenInput[j]);
                }

                // Update layer 1 weights and biases
                for (int j = 0; j < hiddenSize; j++)
                {
                    double gradient = hiddenGradients[j] * input;
                    weightsLayer1[0][j] -= learningRate * gradient;
                    biasesLayer1[j] -= learningRate * hiddenGradients[j];
                }
            }

            double mse = totalError / inputs.Length;
            if (epoch % 100 == 0)
                Console.WriteLine($"Epoch {epoch}: MSE = {mse:F6}");
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

public class NeuralNetworkRegression
{
    private double[][] weightsLayer1;  // input -> hidden
    private double[][] weightsLayer2;  // hidden -> output
    private double[] biasesLayer1;
    private double[] biasesLayer2;
    private int inputSize = 1;
    private int hiddenSize = 16;
    private int outputSize = 1;
    private double learningRate = 0.001;  // Reduced from 0.01
    private Random random = new Random(42);

    // Normalization parameters
    private double xMin, xMax, alphaMin, alphaMax;

    public NeuralNetworkRegression()
    {
        InitializeWeights();
    }

    private void InitializeWeights()
    {
        // Xavier initialization (better than random)
        double limit1 = Math.Sqrt(6.0 / (inputSize + hiddenSize));
        double limit2 = Math.Sqrt(6.0 / (hiddenSize + outputSize));

        weightsLayer1 = new double[inputSize][];
        for (int i = 0; i < inputSize; i++)
        {
            weightsLayer1[i] = new double[hiddenSize];
            for (int j = 0; j < hiddenSize; j++)
                weightsLayer1[i][j] = (random.NextDouble() - 0.5) * 2 * limit1;
        }

        weightsLayer2 = new double[hiddenSize][];
        for (int i = 0; i < hiddenSize; i++)
        {
            weightsLayer2[i] = new double[outputSize];
            for (int j = 0; j < outputSize; j++)
                weightsLayer2[i][j] = (random.NextDouble() - 0.5) * 2 * limit2;
        }

        biasesLayer1 = new double[hiddenSize];
        biasesLayer2 = new double[outputSize];
    }

    private double ReLU(double x) => Math.Max(0, x);
    private double ReLUDerivative(double x) => x > 0 ? 1 : 0;

    private double Normalize(double value, double min, double max)
    {
        if (Math.Abs(max - min) < 1e-10) return 0;
        return (value - min) / (max - min);
    }

    private double Denormalize(double value, double min, double max)
    {
        return value * (max - min) + min;
    }

    public double Predict(double input)
    {
        // Forward pass
        double[] hiddenInput = new double[hiddenSize];
        double[] hiddenOutput = new double[hiddenSize];

        for (int j = 0; j < hiddenSize; j++)
        {
            hiddenInput[j] = weightsLayer1[0][j] * input + biasesLayer1[j];
            hiddenOutput[j] = ReLU(hiddenInput[j]);
        }

        double output = biasesLayer2[0];
        for (int j = 0; j < hiddenSize; j++)
            output += weightsLayer2[j][0] * hiddenOutput[j];

        return output;
    }

    public void Train(double[] inputs, double[] targets, int epochs)
    {
        // Check for NaN in input data
        if (inputs.Any(x => double.IsNaN(x) || double.IsInfinity(x)))
        {
            Console.WriteLine("ERROR: Input data contains NaN or Infinity!");
            return;
        }
        if (targets.Any(x => double.IsNaN(x) || double.IsInfinity(x)))
        {
            Console.WriteLine("ERROR: Target data contains NaN or Infinity!");
            return;
        }

        Console.WriteLine($"Training data size: {inputs.Length}");
        Console.WriteLine($"X range: [{inputs.Min():F4}, {inputs.Max():F4}]");
        Console.WriteLine($"Alpha range: [{targets.Min():F4}, {targets.Max():F4}]");

        for (int epoch = 0; epoch < epochs; epoch++)
        {
            double totalError = 0;
            int validSamples = 0;

            for (int sampleIdx = 0; sampleIdx < inputs.Length; sampleIdx++)
            {
                double input = inputs[sampleIdx];
                double target = targets[sampleIdx];

                // Skip invalid data
                if (double.IsNaN(input) || double.IsNaN(target) || 
                    double.IsInfinity(input) || double.IsInfinity(target))
                    continue;

                validSamples++;

                // Forward pass
                double[] hiddenInput = new double[hiddenSize];
                double[] hiddenOutput = new double[hiddenSize];

                for (int j = 0; j < hiddenSize; j++)
                {
                    hiddenInput[j] = weightsLayer1[0][j] * input + biasesLayer1[j];
                    hiddenOutput[j] = ReLU(hiddenInput[j]);
                }

                double prediction = biasesLayer2[0];
                for (int j = 0; j < hiddenSize; j++)
                    prediction += weightsLayer2[j][0] * hiddenOutput[j];

                // Compute error
                double error = target - prediction;
                
                // Check for NaN
                if (double.IsNaN(error) || double.IsInfinity(error))
                {
                    Console.WriteLine($"NaN detected at sample {sampleIdx}: input={input}, target={target}, prediction={prediction}");
                    continue;
                }

                totalError += error * error;

                // Backpropagation
                double outputGradient = -2 * error;

                // Update layer 2
                for (int j = 0; j < hiddenSize; j++)
                {
                    double gradient = outputGradient * hiddenOutput[j];
                    weightsLayer2[j][0] -= learningRate * gradient;
                }
                biasesLayer2[0] -= learningRate * outputGradient;

                // Update layer 1
                double[] hiddenGradients = new double[hiddenSize];
                for (int j = 0; j < hiddenSize; j++)
                {
                    hiddenGradients[j] = outputGradient * weightsLayer2[j][0] * ReLUDerivative(hiddenInput[j]);
                }

                for (int j = 0; j < hiddenSize; j++)
                {
                    double gradient = hiddenGradients[j] * input;
                    weightsLayer1[0][j] -= learningRate * gradient;
                    biasesLayer1[j] -= learningRate * hiddenGradients[j];
                }
            }

            double mse = validSamples > 0 ? totalError / validSamples : double.NaN;
            
            if (epoch % 100 == 0)
                Console.WriteLine($"Epoch {epoch}: MSE = {mse:F6}, Valid samples = {validSamples}");

            // Detect divergence
            if (double.IsNaN(mse) || mse > 1e6)
            {
                Console.WriteLine("ERROR: Training diverged! Reduce learning rate or check data.");
                break;
            }
        }
    }
}
```
